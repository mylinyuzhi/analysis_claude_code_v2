## 1. Overview: Task ↔ Reminder Integration

The task system integrates with the system reminder module through attachment generation. Task status changes and progress updates are conveyed to the LLM via the attachment pipeline.


## 3. Attachment Producer Functions

### 3.1 Task Status Producer

```javascript
// ============================================
// Task status attachment producer
// Location: chunks.147.mjs
// ============================================

// READABLE (for understanding):
async function produceTaskStatusAttachment(context) {
    const taskManager = getTaskManager();

    // Check if there are pending status changes
    const pendingNotifications = context.pendingTaskNotifications;
    if (!pendingNotifications?.length) {
        return [];
    }

    return pendingNotifications.map(notification => ({
        type: "task_status",
        taskId: notification.taskId,
        previousStatus: notification.previousStatus,
        newStatus: notification.newStatus,
        owner: notification.owner,
        subject: notification.subject,
        timestamp: new Date().toISOString()
    }));
}
```

### 3.2 Task Reminder Producer (Throttled)

```javascript
// ============================================
// Task reminder attachment producer
// Location: chunks.147.mjs
// ============================================

// READABLE (for understanding):
async function produceTaskReminderAttachment(messages, context) {
    // Check if task system is enabled
    if (!isTaskSystemEnabled()) {
        return [];
    }

    // Throttling: only produce every N turns
    const turnCount = countTurnsSinceTaskReminder(messages);
    const TURNS_BETWEEN_REMINDERS = 5;

    if (turnCount < TURNS_BETWEEN_REMINDERS) {
        return [];
    }

    // Load active tasks
    const taskManager = getTaskManager();
    const allTasks = await loadAllTasks(taskManager);

    // Filter to relevant tasks
    const activeTasks = allTasks.filter(t =>
        t.status === "in_progress" || t.status === "pending"
    );

    if (activeTasks.length === 0) {
        return [];
    }

    return [{
        type: "task_reminder",
        activeTasks: activeTasks.map(t => ({
            id: t.id,
            subject: t.subject,
            status: t.status,
            owner: t.owner,
            blockedBy: t.blockedBy
        })),
        completedCount: allTasks.filter(t => t.status === "completed").length,
        pendingCount: allTasks.filter(t => t.status === "pending").length,
        inProgressCount: allTasks.filter(t => t.status === "in_progress").length,
        turnCount: turnCount
    }];
}
```

## 4. Dependency Resolution → Reminder Flow

When a task completes, its dependencies may become unblocked:

```
Task [1] completes (status: "completed")
    │
    ├─→ TaskCompleted hooks execute (Hi6)
    │     └─→ Validation passes
    │
    ├─→ Update dependent tasks' blockedBy arrays
    │     └─→ Remove "1" from blockedBy of task [2], [3]
    │
    └─→ Generate dependency_resolved attachments
          │
          ├─→ For task [2]: "Dependency [1] resolved, you can now proceed"
          │
          └─→ For task [3]: "Dependency [1] resolved, still blocked by [2]"
```

### Dependency Resolution Logic

```javascript
// ============================================
// Dependency resolution on task completion
// Location: chunks.84.mjs (updateTask)
// ============================================

// READABLE (for understanding):
async function resolveDependenciesOnCompletion(taskManager, completedTaskId) {
    const allTasks = await loadAllTasks(taskManager);

    // Find tasks that were blocked by this task
    const dependentTasks = allTasks.filter(t =>
        t.blockedBy.includes(completedTaskId)
    );

    const notifications = [];

    for (const dependent of dependentTasks) {
        // Remove completed task from blockedBy
        const newBlockedBy = dependent.blockedBy.filter(id => id !== completedTaskId);

        // Update the task
        await updateTask(taskManager, dependent.id, {
            blockedBy: newBlockedBy
        });

        // Check if fully unblocked
        const isNowUnblocked = newBlockedBy.length === 0;

        notifications.push({
            type: "dependency_resolved",
            taskId: dependent.id,
            resolvedTaskId: completedTaskId,
            isFullyUnblocked: isNowUnblocked,
            remainingBlockers: newBlockedBy
        });
    }

    return notifications;
}
```

## 5. Claim Operation → Owner Assignment

When a task is claimed in team mode, the owner is assigned and reflected in reminders:

```javascript
// ============================================
// OT8 - claimTask with owner assignment
// Location: chunks.84.mjs:1781
// ============================================

// ORIGINAL (for source lookup):
async function OT8(A, q, K, Y = {}) {
    let z = await DB(A, q);
    if (!z) throw Error(`Task ${q} not found`);
    if (z.status !== "pending") throw Error(`Task ${q} is not pending (status: ${z.status})`);
    if (z.owner) throw Error(`Task ${q} is already claimed by ${z.owner}`);

    // Claim the task
    let _ = await WI(A, q, {
        status: "in_progress",
        owner: K  // Assign owner
    }, Y);

    return _
}

// READABLE (for understanding):
async function claimTask(taskManager, taskId, agentName, options = {}) {
    // Load and validate
    let task = await loadTask(taskManager, taskId);
    if (!task) {
        throw new Error(`Task ${taskId} not found`);
    }
    if (task.status !== "pending") {
        throw new Error(`Task ${taskId} is not pending (status: ${task.status})`);
    }
    if (task.owner) {
        throw new Error(`Task ${taskId} is already claimed by ${task.owner}`);
    }

    // Claim the task (sets owner and status)
    let updatedTask = await updateTask(taskManager, taskId, {
        status: "in_progress",
        owner: agentName  // Assign owner for team coordination
    }, options);

    // Attachment will include owner in notification
    return updatedTask;
}

// Mapping: OT8→claimTask, DB→loadTask, WI→updateTask, A→taskManager, q→taskId, K→agentName
```

## 6. TodoWrite Integration

The TodoWrite tool integrates with the reminder system for simple task tracking:

```javascript
// ============================================
// TodoWrite tool attachment generation
// Location: chunks.141.mjs (TodoWrite tool)
// ============================================

// READABLE (for understanding):
const TodoWriteTool = {
    name: "TodoWrite",

    async call(input, context) {
        const { todos } = input;

        // Validate todos
        for (const todo of todos) {
            if (!todo.content || !todo.status) {
                throw new Error("Each todo must have content and status");
            }
        }

        // Store in session state
        context.setAppState(state => ({
            ...state,
            todos: todos
        }));

        // Generate todo attachment for LLM context
        return {
            data: { todos },
            attachment: {
                type: "todo",
                todos: todos
            }
        };
    },

    // Render result message
    mapToolResultToToolResultBlockParam(result, toolUseId) {
        const activeCount = result.todos.filter(t => t.status === "in_progress").length;
        const pendingCount = result.todos.filter(t => t.status === "pending").length;
        const completedCount = result.todos.filter(t => t.status === "completed").length;

        return {
            type: "tool_result",
            content: `Todo list updated:\n` +
                     `  • ${activeCount} in progress\n` +
                     `  • ${pendingCount} pending\n` +
                     `  • ${completedCount} completed`,
            tool_use_id: toolUseId
        };
    }
};
```

## 8. Cross-Module Integration Summary

### Task System → System Reminder

| Event | Attachment Type | Trigger |
|-------|-----------------|---------|
| Task created | `task_status` | TaskCreate tool call |
| Status changed | `task_status` | TaskUpdate tool call |
| Owner assigned | `task_status` | claimTask operation |
| Dependency resolved | `dependency_resolved` | Task completion |
| Periodic reminder | `task_reminder` | Throttled (every 5 turns) |
| Todo updated | `todo` | TodoWrite tool call |
| Todo reminder | `todo_reminder` | Throttled (every 3-5 turns) |

### System Reminder → Task System

| Attachment Type | Consumer | Effect |
|-----------------|----------|--------|
| `task_reminder` | LLM | Context about active tasks |
| `todo_reminder` | LLM | Context about todo list |
| `task_status` | LLM | Notification of state changes |
